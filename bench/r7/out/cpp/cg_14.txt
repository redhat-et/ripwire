304	std::string finalSegment( std::string_view raw )   // allocates a std::string → not noexcept
305	{
306	    // A C++ OPERATOR name (`operator<`, `operator<<`, `operator<=>`, `operator bool`) is the one place a
307	    // name legitimately carries `<`/`::`-looking punctuation that is NOT a scope/type-arg separator — the
308	    // `<`-strip and "::"/"." splits below would mangle it to a bare `operator`. Detect the `operator`
309	    // keyword segment (bare `operator...` or the last `Scope::operator...`) and return it whole. Guarded on
310	    // a following non-identifier char so a normal identifier that merely starts with "operator" (e.g.
311	    // `operatorId`) still falls through to the generic path unchanged.
312	    if( const std::size_t op = raw.rfind( "operator" ); op != std::string_view::npos )
313	    {
314	        const bool atSegStart = ( op == 0 ) || ( raw[ op - 1 ] == ':' ) || ( raw[ op - 1 ] == '.' );
315	        const std::size_t after = op + 8;   // one-past `operator`
316	        const bool isOpToken = after >= raw.size()
317	                            || !( std::isalnum( static_cast<unsigned char>( raw[ after ] ) ) || raw[ after ] == '_' );
318	        if( atSegStart && isOpToken )
319	        {
320	            return std::string( raw.substr( op ) );   // `operator<<`, `operator bool`, etc. — verbatim
321	        }
322	    }
323	
324	    // FIX #1 (generic base/impl): drop a type-argument list before ANYTHING else. A base/impl name
325	    // handed back as `generic_type` carries `<...>` (Java `Base<String>`, Rust `Wrapper<T>`) whose
326	    // bare form (`Base`/`Wrapper`) is what byName keys on. The strip MUST precede the "::"/"." split,
327	    // or a `::` INSIDE the args (`Foo<A::B>`) would be mistaken for the segment separator. A name never
328	    // legitimately contains a bare '<' (only a type-argument list opens one), so truncating at the FIRST
329	    // '<' is safe; the C++/TS bare-identifier path has no '<' → no-op (byte-identical).
330	    if( const std::size_t lt = raw.find( '<' ); lt != std::string_view::npos )
331	    {
332	        raw = raw.substr( 0, lt );
333	    }
334	
335	    // already a bare identifier in nearly all cases; defensive split on '.' and "::".
336	    std::size_t pos = raw.rfind( "::" );
337	    if( pos != std::string_view::npos )
338	    {
339	        raw = raw.substr( pos + 2 );
340	    }
341	    pos = raw.rfind( '.' );
342	    if( pos != std::string_view::npos )
343	    {
344	        raw = raw.substr( pos + 1 );
345	    }
346	
347	    // trim trailing whitespace a `Base <String>` / `Wrapper <T>` spacing left behind (grammars usually
348	    // hand back tight text, but a space before the '<' would otherwise leave `Base ` after the strip).
349	    while( !raw.empty() && ( raw.back() == ' ' || raw.back() == '\t' ) )
350	    {
351	        raw.remove_suffix( 1 );
352	    }
353	    return std::string( raw );

577	        }
578	        else if( p.b == i && !landed[p.a] )
579	        {

581	        }

594	        {
