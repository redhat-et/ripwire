271	// ── Chilimbi's separation weight, PLDI 1999 §4 ───────────────────────────────────────────────────────
272	// wt(fi,fj) = ( cache_block_size - dist(fi,fj) ) / cache_block_size, dist = bytes between field starts.
273	// Clamped at zero: past one block the pair simply cannot share a line and the linear form would go
274	// negative. This is a CITED formula, reproduced — not a ripwire invention. Do not "improve" it without
275	// saying so in the output, because the number is comparable to a published one only while it is his.
276	inline double separationWeight( std::uint32_t distBytes ) noexcept
277	{
278	    if( distBytes >= kCacheBlockBytes )
279	    {
280	        return 0.0;
281	    }
282	    return double( kCacheBlockBytes - distBytes ) / double( kCacheBlockBytes );
283	}

285	// A field straddles a line when its first and last byte fall in different 64 B blocks. A field LARGER
286	// than a block always spans two and is not a defect — it cannot do otherwise — so it is excluded.
287	inline bool straddlesLine( std::uint32_t offset, std::uint32_t size ) noexcept
288	{
289	    if( size == 0 || size > kCacheBlockBytes )
290	    {
291	        return false;
292	    }
293	    return ( offset / kCacheBlockBytes ) != ( ( offset + size - 1 ) / kCacheBlockBytes );
294	}
