157	// The risk profile of a whole ingested tree. Deliberately NOT keyed by symbol identity: DMM compares two
158	// VOLUMES, and an unchanged unit lands in the same bin with the same volume on both sides, so it cancels to
159	// zero without ever being matched. That is also what makes a pure file move free — the volume leaves one
160	// path and arrives at another inside the same sum.
161	inline SideProfile profileOf( const IngestResult& ing )
162	{
163	    SideProfile out;
164	    for( const Symbol& s : ing.symbols )
165	    {
166	        if( !isUnit( s ) )
167	        {
168	            continue;
169	        }
170	        ++out.unitCount;
171	        out.volume += s.loc;
172	        for( std::size_t propIndex = 0; propIndex < kPropCount; ++propIndex )
173	        {
174	            RiskProfile& p = out.byProp[propIndex];
175	            if( isLowRisk( s, propIndex ) )
176	            {
177	                p.lowVolume += s.loc;
178	            }
179	            else
180	            {
181	                p.highVolume += s.loc;
182	            }
183	        }
184	    }
185	    return out;
