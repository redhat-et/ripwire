25	 * map.addAll(new Map([["c", 3], ["d", 4]]), true);
26	 * map.get("a"); // 1
27	 * map.get("d"); // 4
28	 * for (const [key, value] of map) {

30	 * }
31	 * ```

33	 * @template V
